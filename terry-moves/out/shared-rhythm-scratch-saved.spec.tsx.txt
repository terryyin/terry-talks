import { createHash } from 'node:crypto';
import { readFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { registeredRhythm, rhythmPicture } from './renderedFilm';
import { SharedRhythmFilmProps } from '../../src/musicAndSound/SharedRhythmFilm';

test('an explicitly saved added passage runs through the actual registered common composition', () => {
	const props: SharedRhythmFilmProps = JSON.parse(readFileSync(resolve(__dirname, '../../out/shared-rhythm-scratch-props.json'), 'utf8'));
	const entry = registeredRhythm('SharedRhythmRevision');
	expect(entry.calculateMetadata({ props })).toEqual({ durationInFrames: 1116 });
	const page = rhythmPicture('SharedRhythmRevision', 1089, props);
	expect(page.querySelector('[data-rhythm-passage]')!.getAttribute('data-rhythm-passage')).toBe('scratch-answer');
	expect(page.querySelectorAll('[data-rhythm-performer]')).toHaveLength(1);
	const answer = page.querySelector('[data-rhythm-performer="answer"]')!;
	expect(answer.getAttribute('data-rhythm-impact')).toBe('scratch-pluck');
	expect(answer.getAttribute('transform')).toBe('translate(540 560) scale(1.2 0.84)');
	expect(answer.querySelector('rect')!.getAttribute('width')).toBe('180');
	expect(page.querySelector('audio')!.getAttribute('src')).toContain('/assets/shared-rhythm/scratch.wav');
	const wav = readFileSync(resolve(__dirname, '../../public', props.score.audioPath));
	expect(createHash('sha256').update(wav).digest('hex')).toBe(props.score.wavSha256);
	const begin = Math.round(1089 / 30 * 44100);
	let onset = begin;
	while (onset < begin + 44100 / 30 && Math.abs(wav.readInt16LE(44 + onset * 2)) <= 100) onset++;
	expect(onset / 44100 - 1089 / 30).toBeLessThan(1 / 30);
});
